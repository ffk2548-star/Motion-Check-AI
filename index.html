import os import sys import subprocess # ------------------------------------------------------------- # 1. Automatic Package Installer (รันและติดตั้ง Library อัตโนมัติ) # ------------------------------------------------------------- REQUIRED_PACKAGES = { "streamlit": "streamlit>=1.30.0", "mediapipe": "mediapipe>=0.10.9", "cv2": "opencv-python-headless>=4.8.0", "numpy": "numpy>=1.24.0", "PIL": "pillow>=10.0.0" } def install_missing_packages(): for module_name, package_spec in REQUIRED_PACKAGES.items(): try: __import__(module_name) except ImportError: subprocess.check_call([sys.executable, "-m", "pip", "install", package_spec]) install_missing_packages() # ------------------------------------------------------------- # 2. Import Libraries # ------------------------------------------------------------- import cv2 import numpy as np import mediapipe as mp import streamlit as st # ------------------------------------------------------------- # 3. Streamlit Page UI Configuration # ------------------------------------------------------------- st.set_page_config( page_title="AI Exercise Tracker & Biomechanics", page_icon="🏋️‍♂️", layout="wide" ) st.title("🏋️‍♂️ AI Motion Tracker & Biomechanics Analyser") st.markdown("ระบบวิเคราะห์และตรวจจับท่าทางการออกกำลังกายแบบ Real-time ตามหลักชีวกลศาสตร์") # Sidebar Configuration st.sidebar.header("⚙️ ตั้งค่าการตรวจจับ") exercise_mode = st.sidebar.selectbox( "เลือกท่าทางที่ต้องการฝึก:", ["Squat", "Leg Lunge", "Push Up", "Plank"] ) run_app = st.sidebar.checkbox("เปิดกล้อง Webcam", value=False) st.sidebar.markdown("---") st.sidebar.subheader("📐 เกณฑ์มุมทางชีวกลศาสตร์") if exercise_mode == "Squat": st.sidebar.info("• มุมเข่าต่ำสุด (Down): ≤ 100°\n• มุมเข่ายืนตรง (Up): ≥ 160°") elif exercise_mode == "Leg Lunge": st.sidebar.info("• มุมเข่าขาหน้า (Down): ≤ 95°\n• มุมเข่ายืนตรง (Up): ≥ 160°") elif exercise_mode == "Push Up": st.sidebar.info("• มุมข้อศอกหย่อนตัว (Down): ≤ 90°\n• มุมข้อศอกดันขึ้น (Up): ≥ 160°") elif exercise_mode == "Plank": st.sidebar.info("• มุมลำตัวตรง (Neutral Spine): 165° - 180°") # ------------------------------------------------------------- # 4. Biomechanics Logic & Mathematics # ------------------------------------------------------------- def calculate_angle(a, b, c): """คำนวณมุมระหว่าง 3 จุด (a=จุดเริ่มต้น, b=จุดหมุน, c=จุดปลาย)""" a = np.array(a) b = np.array(b) c = np.array(c) radians = np.arctan2(c[1]-b[1], c[0]-b[0]) - np.arctan2(a[1]-b[1], a[0]-b[0]) angle = np.abs(radians * 180.0 / np.pi) if angle > 180.0: angle = 360.0 - angle return int(angle) mp_pose = mp.solutions.pose mp_drawing = mp.solutions.drawing_utils # ------------------------------------------------------------- # 5. Dashboard Layout & Processing Loop # ------------------------------------------------------------- col1, col2 = st.columns([3, 1]) with col2: st.metric(label="ท่าที่กำลังตรวจจับ", value=exercise_mode) counter_placeholder = st.empty() status_placeholder = st.empty() angle_placeholder = st.empty() with col1: FRAME_WINDOW = st.image([]) if run_app: cap = cv2.VideoCapture(0) counter = 0 stage = None with mp_pose.Pose(min_detection_confidence=0.7, min_tracking_confidence=0.7) as pose: while cap.isOpened() and run_app: ret, frame = cap.read() if not ret: st.error("ไม่สามารถเชื่อมต่อกล้อง Webcam ได้") break # Process Image Frame frame = cv2.flip(frame, 1) image = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) results = pose.process(image) current_angle = 0 feedback = "พร้อมปฏิบัติ" if results.pose_landmarks: landmarks = results.pose_landmarks.landmark # Keypoint Extraction shoulder = [landmarks[mp_pose.PoseLandmark.LEFT_SHOULDER.value].x, landmarks[mp_pose.PoseLandmark.LEFT_SHOULDER.value].y] elbow = [landmarks[mp_pose.PoseLandmark.LEFT_ELBOW.value].x, landmarks[mp_pose.PoseLandmark.LEFT_ELBOW.value].y] wrist = [landmarks[mp_pose.PoseLandmark.LEFT_WRIST.value].x, landmarks[mp_pose.PoseLandmark.LEFT_WRIST.value].y] hip = [landmarks[mp_pose.PoseLandmark.LEFT_HIP.value].x, landmarks[mp_pose.PoseLandmark.LEFT_HIP.value].y] knee = [landmarks[mp_pose.PoseLandmark.LEFT_KNEE.value].x, landmarks[mp_pose.PoseLandmark.LEFT_KNEE.value].y] ankle = [landmarks[mp_pose.PoseLandmark.LEFT_ANKLE.value].x, landmarks[mp_pose.PoseLandmark.LEFT_ANKLE.value].y] # Biomechanics Analysis if exercise_mode == "Squat": current_angle = calculate_angle(hip, knee, ankle) if current_angle <= 100: stage = "down" feedback = "ย่อตัวได้ระดับแล้ว" if current_angle >= 160 and stage == "down": stage = "up" counter += 1 feedback = "นับ 1 ครั้ง!" elif exercise_mode == "Leg Lunge": current_angle = calculate_angle(hip, knee, ankle) if current_angle <= 95: stage = "down" feedback = "เข่าทำมุม 90 องศาแล้ว" if current_angle >= 160 and stage == "down": stage = "up" counter += 1 feedback = "นับ 1 ครั้ง!" elif exercise_mode == "Push Up": current_angle = calculate_angle(shoulder, elbow, wrist) if current_angle <= 90: stage = "down" feedback = "หย่อนอกลงได้ระดับแล้ว" if current_angle >= 160 and stage == "down": stage = "up" counter += 1 feedback = "นับ 1 ครั้ง!" elif exercise_mode == "Plank": current_angle = calculate_angle(shoulder, hip, ankle) if 165 <= current_angle <= 180: stage = "Good Form" feedback = "เกร็งลำตัวตรงได้ดี" else: stage = "Adjust Spine" feedback = "ปรับระดับสะโพก/หลังให้ตรง!" # Render Skeletal Overlay mp_drawing.draw_landmarks( image, results.pose_landmarks, mp_pose.POSE_CONNECTIONS ) # Update Interface Elements counter_placeholder.metric( label="จำนวนครั้ง (Reps)" if exercise_mode != "Plank" else "สถานะท่าทาง", value=counter if exercise_mode != "Plank" else (stage if stage else "-") ) status_placeholder.markdown(f"**คำแนะนำ:** {feedback}") angle_placeholder.markdown(f"**มุมข้อต่อปัจจุบัน:** `{current_angle}°`") FRAME_WINDOW.image(image) cap.release() else: st.info("👈 กรุณากดติ๊กถูกที่ช่อง 'เปิดกล้อง Webcam' บนเมนูด้านซ้ายเพื่อเริ่มใช้งาน")
